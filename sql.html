Common SQL Commands

These are some of the most common commands you will see when reading or writing SQL.

SELECT

Used to read data from a table.

SELECT name, age FROM users;

FROM

Tells SQL which table to pull the data from.

SELECT * FROM orders;

WHERE

Filters rows so you only get the ones that match a condition.

SELECT * FROM users WHERE age > 21;

ORDER BY

Sorts the results in ascending or descending order.

SELECT * FROM scores ORDER BY points DESC;

GROUP BY

Groups rows together so you can aggregate them.

SELECT department, COUNT(*) FROM employees GROUP BY department;

HAVING

Filters grouped results after a GROUP BY.

SELECT department, COUNT(*) FROM employees GROUP BY department HAVING COUNT(*) > 5;

INSERT INTO

Adds a new row into a table.

INSERT INTO users (name, age) VALUES ('Ashton', 22);

UPDATE

Changes existing data in a table.

UPDATE users SET age = 23 WHERE name = 'Ashton';

DELETE

Removes rows from a table.

DELETE FROM users WHERE age < 18;

JOIN

Combines rows from two tables based on a related column.

SELECT users.name, orders.total FROM users JOIN orders ON users.id = orders.user_id;

COUNT

Returns how many rows match something.

SELECT COUNT(*) FROM users;

SUM

Adds up values in a numeric column.

SELECT SUM(price) FROM orders;

AVG

Returns the average value of a numeric column.

SELECT AVG(score) FROM exams;

MAX and MIN

Return the largest or smallest value in a column.

SELECT MAX(score), MIN(score) FROM exams;

LIMIT

Restricts how many rows are returned.

SELECT * FROM users LIMIT 5;
